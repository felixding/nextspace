cmake_minimum_required(VERSION 3.19)
project(NextSpaceNativeUI LANGUAGES C OBJC)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr/NextSpace" CACHE PATH "NEXTSPACE prefix" FORCE)
endif()
set(CMAKE_BUILD_RPATH "$ORIGIN;/usr/NextSpace/lib;/Library/Libraries")
set(CMAKE_INSTALL_RPATH "$ORIGIN;/usr/NextSpace/lib;/Library/Libraries")
find_program(GNUSTEP_CONFIG gnustep-config HINTS /usr/NextSpace/bin REQUIRED)
execute_process(COMMAND ${GNUSTEP_CONFIG} --objc-flags OUTPUT_VARIABLE OBJC_FLAGS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND ${GNUSTEP_CONFIG} --gui-libs OUTPUT_VARIABLE GUI_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(OBJC_FLAGS UNIX_COMMAND "${OBJC_FLAGS}")
list(REMOVE_ITEM OBJC_FLAGS "-MMD" "-MP")
separate_arguments(GUI_LIBS UNIX_COMMAND "${GUI_LIBS}")
add_library(nextspace-native-ui SHARED Loader.c)
target_include_directories(nextspace-native-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nextspace-native-ui PRIVATE dl pthread)
set_target_properties(nextspace-native-ui PROPERTIES VERSION 0.1 SOVERSION 1 AUTOMOC OFF)
add_library(nextspace-native-ui-gnustep SHARED GNUstep/Provider.m GNUstep/ViewHost.m GNUstep/EventSource.m GNUstep/Scroller.m GNUstep/Controls.m GNUstep/Menu.m)
target_compile_options(nextspace-native-ui-gnustep PRIVATE ${OBJC_FLAGS} -fvisibility=hidden)
target_link_libraries(nextspace-native-ui-gnustep PRIVATE ${GUI_LIBS})
set_target_properties(nextspace-native-ui-gnustep PROPERTIES VERSION 0.1 SOVERSION 1 AUTOMOC OFF)
install(TARGETS nextspace-native-ui nextspace-native-ui-gnustep LIBRARY DESTINATION lib/nextspace-scrollbars)
install(FILES NativeUI.h DESTINATION include/NextSpaceNativeUI)
add_executable(native-controls-probe ../../Tests/ForeignScrollers/native-controls-probe.c)
target_link_libraries(native-controls-probe PRIVATE nextspace-native-ui)
target_compile_options(native-controls-probe PRIVATE -UNDEBUG)
add_dependencies(native-controls-probe nextspace-native-ui-gnustep)
set_target_properties(native-controls-probe PROPERTIES AUTOMOC OFF)
