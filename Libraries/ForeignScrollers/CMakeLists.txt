cmake_minimum_required(VERSION 3.19)
project(NextSpaceForeignScrollers LANGUAGES C CXX OBJC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_BUILD_RPATH "/usr/NextSpace/lib;/Library/Libraries")
set(CMAKE_INSTALL_RPATH "$ORIGIN;$ORIGIN/../..;/usr/NextSpace/lib;/Library/Libraries")
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr/NextSpace" CACHE PATH "NEXTSPACE prefix" FORCE)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
find_program(GNUSTEP_CONFIG gnustep-config HINTS /usr/NextSpace/bin REQUIRED)
execute_process(COMMAND ${GNUSTEP_CONFIG} --objc-flags OUTPUT_VARIABLE OBJC_FLAGS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND ${GNUSTEP_CONFIG} --gui-libs OUTPUT_VARIABLE GUI_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(OBJC_FLAGS UNIX_COMMAND "${OBJC_FLAGS}")
list(REMOVE_ITEM OBJC_FLAGS "-MMD" "-MP")
separate_arguments(GUI_LIBS UNIX_COMMAND "${GUI_LIBS}")
add_library(nextspace-native-scroller SHARED NativeScroller.m)
target_compile_options(nextspace-native-scroller PRIVATE ${OBJC_FLAGS})
target_link_libraries(nextspace-native-scroller PRIVATE ${GUI_LIBS})
set_target_properties(nextspace-native-scroller PROPERTIES AUTOMOC OFF INSTALL_RPATH "/usr/NextSpace/lib;/Library/Libraries")
set_target_properties(nextspace-native-scroller PROPERTIES VERSION 0.1 SOVERSION 1)
add_executable(native-scroller-probe ../../Tests/ForeignScrollers/native-probe.c)
target_link_libraries(native-scroller-probe PRIVATE nextspace-native-scroller)
target_include_directories(native-scroller-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(native-scroller-probe PRIVATE -UNDEBUG)
set_target_properties(native-scroller-probe PROPERTIES AUTOMOC OFF)
add_library(nextspace-qt-scrollbars MODULE QtScrollers.cpp)
target_link_libraries(nextspace-qt-scrollbars PRIVATE Qt6::Widgets nextspace-native-scroller)
set_target_properties(nextspace-qt-scrollbars PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
add_library(nextspace-gtk4-scrollbars MODULE Gtk4Scrollers.c)
target_link_libraries(nextspace-gtk4-scrollbars PRIVATE PkgConfig::GTK4 nextspace-native-scroller m)
set_target_properties(nextspace-gtk4-scrollbars PROPERTIES AUTOMOC OFF)
add_library(nextspace-gtk-bootstrap SHARED GtkBootstrap.c)
target_link_libraries(nextspace-gtk-bootstrap PRIVATE dl)
set_target_properties(nextspace-gtk-bootstrap PROPERTIES AUTOMOC OFF)
add_executable(qt-scroller-probe ../../Tests/ForeignScrollers/qt-probe.cpp)
target_link_libraries(qt-scroller-probe PRIVATE Qt6::Widgets)
add_executable(gtk4-scroller-probe ../../Tests/ForeignScrollers/gtk4-probe.c)
target_link_libraries(gtk4-scroller-probe PRIVATE PkgConfig::GTK4 m)
set_target_properties(gtk4-scroller-probe PROPERTIES AUTOMOC OFF)
add_library(nextspace-gtk3-scrollbars MODULE Gtk3Scrollers.c)
target_link_libraries(nextspace-gtk3-scrollbars PRIVATE PkgConfig::GTK3 nextspace-native-scroller m)
set_target_properties(nextspace-gtk3-scrollbars PROPERTIES AUTOMOC OFF)
add_executable(gtk3-scroller-probe ../../Tests/ForeignScrollers/gtk4-probe.c)
target_link_libraries(gtk3-scroller-probe PRIVATE PkgConfig::GTK3 m)
set_target_properties(gtk3-scroller-probe PROPERTIES AUTOMOC OFF)
add_library(nextspace-scroller-inspector MODULE ../../Tests/ForeignScrollers/qt-inspector.cpp)
target_link_libraries(nextspace-scroller-inspector PRIVATE Qt6::Widgets)
set_target_properties(nextspace-scroller-inspector PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
install(TARGETS nextspace-native-scroller nextspace-gtk-bootstrap nextspace-gtk3-scrollbars nextspace-gtk4-scrollbars
  LIBRARY DESTINATION lib/nextspace-scrollbars)
install(TARGETS nextspace-qt-scrollbars LIBRARY DESTINATION lib/nextspace-scrollbars/qt6/generic)
install(FILES session.sh DESTINATION lib/nextspace-scrollbars)
install(PROGRAMS konsole nextspace-native-scrollbars DESTINATION bin)
