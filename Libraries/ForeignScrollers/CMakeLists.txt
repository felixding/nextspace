cmake_minimum_required(VERSION 3.19)
project(NextSpaceForeignScrollers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_RPATH "/usr/NextSpace/lib;/Library/Libraries")
set(CMAKE_INSTALL_RPATH "$ORIGIN;$ORIGIN/../..;/usr/NextSpace/lib;/Library/Libraries")
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr/NextSpace" CACHE PATH "NEXTSPACE prefix" FORCE)
endif()
option(NEXTSPACE_BUILD_TOOLKIT_ADAPTERS "Build the GTK/Qt adapters and their probes" ON)
option(NXS_BUILD_GTK3_GEOMETRY "Build the GTK3 Wayland geometry bridge" ${NEXTSPACE_BUILD_TOOLKIT_ADAPTERS})
add_subdirectory(../NativeUI ${CMAKE_BINARY_DIR}/native-ui)
add_library(nextspace-native-scroller SHARED NativeScroller.c)
target_link_libraries(nextspace-native-scroller PRIVATE nextspace-native-ui)
set_target_properties(nextspace-native-scroller PROPERTIES AUTOMOC OFF
  INSTALL_RPATH "$ORIGIN;/usr/NextSpace/lib;/Library/Libraries" VERSION 0.1 SOVERSION 1)
add_executable(native-scroller-probe ../../Tests/ForeignScrollers/native-probe.c)
target_link_libraries(native-scroller-probe PRIVATE nextspace-native-scroller)
target_include_directories(native-scroller-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(native-scroller-probe PRIVATE -UNDEBUG)
install(TARGETS nextspace-native-scroller LIBRARY DESTINATION lib/nextspace-scrollbars)
install(PROGRAMS ../NativeUI/Host/nextspace-native-ui-host DESTINATION bin)
install(FILES ../NativeUI/Host/main.py ../NativeUI/Host/menu_service.py ../NativeUI/Host/connections.py DESTINATION libexec/nextspace-native-ui)
install(FILES ../../Experiments/ChromiumScrollers/helper.py ../../Experiments/ChromiumScrollers/native_controls.py
  ../../Experiments/ChromiumScrollers/native_menus.py ../NativeUI/Bridge/color_owners.py ../NativeUI/Bridge/action_log.py ../NativeUI/Bridge/auxiliary_activation.py DESTINATION libexec/nextspace-native-ui/ChromiumScrollers)
install(FILES ../../Experiments/DBusMenus/binding.py ../../Experiments/DBusMenus/broker.py
  ../../Experiments/DBusMenus/glib_loop.py ../../Experiments/DBusMenus/menu_model.py ../../Experiments/DBusMenus/menu_events.py
  DESTINATION libexec/nextspace-native-ui/DBusMenus)
if(NEXTSPACE_BUILD_TOOLKIT_ADAPTERS)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0 gio-2.0)
  pkg_check_modules(GTK3_MENUS REQUIRED IMPORTED_TARGET dbusmenu-gtk3-0.4 wayland-client)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  set(APPMENU_XML "${CMAKE_CURRENT_SOURCE_DIR}/../../Protocols/appmenu.xml")
  set(APPMENU_HEADER "${CMAKE_CURRENT_BINARY_DIR}/appmenu-client-protocol.h")
  set(APPMENU_CODE "${CMAKE_CURRENT_BINARY_DIR}/appmenu-protocol.c")
  add_custom_command(OUTPUT "${APPMENU_HEADER}" "${APPMENU_CODE}"
    COMMAND "${WAYLAND_SCANNER}" client-header "${APPMENU_XML}" "${APPMENU_HEADER}"
    COMMAND "${WAYLAND_SCANNER}" private-code "${APPMENU_XML}" "${APPMENU_CODE}"
    DEPENDS "${APPMENU_XML}" VERBATIM)
  set(CMAKE_AUTOMOC ON)
  add_library(nextspace-qt-scrollbars MODULE QtScrollers.cpp)
  target_link_libraries(nextspace-qt-scrollbars PRIVATE Qt6::Widgets nextspace-native-scroller nextspace-native-ui)
  set_target_properties(nextspace-qt-scrollbars PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
  add_library(nextspace-gtk4-scrollbars MODULE Gtk4Scrollers.c)
  target_link_libraries(nextspace-gtk4-scrollbars PRIVATE PkgConfig::GTK4 nextspace-native-scroller m)
  set_target_properties(nextspace-gtk4-scrollbars PROPERTIES AUTOMOC OFF)
  add_library(nextspace-gtk-bootstrap SHARED GtkBootstrap.c)
  target_link_libraries(nextspace-gtk-bootstrap PRIVATE dl)
  set_target_properties(nextspace-gtk-bootstrap PROPERTIES AUTOMOC OFF)
  add_executable(qt-scroller-probe ../../Tests/ForeignScrollers/qt-probe.cpp)
  target_link_libraries(qt-scroller-probe PRIVATE Qt6::Widgets)
  add_executable(gtk4-scroller-probe ../../Tests/ForeignScrollers/gtk4-probe.c)
  target_link_libraries(gtk4-scroller-probe PRIVATE PkgConfig::GTK4 m)
  set_target_properties(gtk4-scroller-probe PROPERTIES AUTOMOC OFF)
  add_library(nextspace-gtk3-scrollbars MODULE Gtk3Scrollers.c Gtk3Controls.c Gtk3Editing.c Gtk3AppMenu.c Gtk3Context.c
    "${APPMENU_HEADER}" "${APPMENU_CODE}")
  target_include_directories(nextspace-gtk3-scrollbars PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(nextspace-gtk3-scrollbars PRIVATE
    PkgConfig::GTK3 PkgConfig::GTK3_MENUS nextspace-native-scroller nextspace-native-ui m)
  set_target_properties(nextspace-gtk3-scrollbars PROPERTIES AUTOMOC OFF)
  add_executable(gtk3-scroller-probe ../../Tests/ForeignScrollers/gtk4-probe.c)
  target_link_libraries(gtk3-scroller-probe PRIVATE PkgConfig::GTK3 m)
  set_target_properties(gtk3-scroller-probe PROPERTIES AUTOMOC OFF)
  add_library(nextspace-scroller-inspector MODULE ../../Tests/ForeignScrollers/qt-inspector.cpp)
  target_link_libraries(nextspace-scroller-inspector PRIVATE Qt6::Widgets)
  set_target_properties(nextspace-scroller-inspector PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
  install(TARGETS nextspace-gtk-bootstrap nextspace-gtk3-scrollbars nextspace-gtk4-scrollbars
    LIBRARY DESTINATION lib/nextspace-scrollbars)
  install(TARGETS nextspace-qt-scrollbars LIBRARY DESTINATION lib/nextspace-scrollbars/qt6/generic)
  install(FILES session.sh DESTINATION lib/nextspace-scrollbars)
  install(PROGRAMS konsole nextspace-native-scrollbars DESTINATION bin)
endif()
